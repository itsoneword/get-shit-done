```javascript
/**
 * agent-spec-init — Tests for AGENT-SPEC.md discovery in init.cjs
 */
const fs = require('fs');
const path = require('path');
const os = require('os');
const { execSync } = require('child_process');

const GSD_TOOLS = path.join(__dirname, '..', 'get-shit-done', 'bin', 'gsd-tools.cjs');

function runInit(cwd) {
  const raw = execSync(`node "${GSD_TOOLS}" init plan-phase 01`, {
    cwd,
    encoding: 'utf-8',
    timeout: 10000,
  });
  if (raw.startsWith('@file:')) {
    return JSON.parse(fs.readFileSync(raw.replace('@file:', '').trim(), 'utf-8'));
  }
  return JSON.parse(raw);
}

function setupProject(tmpDir, opts = {}) {
  // Minimal .planning structure
  const planningDir = path.join(tmpDir, '.planning');
  const phaseDir = path.join(planningDir, 'phases', '01-test-phase');
  fs.mkdirSync(phaseDir, { recursive: true });

  // STATE.md
  fs.writeFileSync(path.join(planningDir, 'STATE.md'), [
    '---',
    'milestone: v1.0',
    'status: active',
    '---',
    '## Current Position',
    'Phase: 01 (test-phase)',
  ].join('\n'));

  // ROADMAP.md
  fs.writeFileSync(path.join(planningDir, 'ROADMAP.md'), [
    '# Roadmap: Test v1.0',
    '## Phases',
    '- [ ] **Phase 1: Test Phase** - Test',
    '## Phase Details',
    '### Phase 1: Test Phase',
    '**Goal**: Test goal',
    '**Depends on**: Nothing',
    '**Requirements**: TEST-01',
    '**Plans**: TBD',
  ].join('\n'));

  // REQUIREMENTS.md
  fs.writeFileSync(path.join(planningDir, 'REQUIREMENTS.md'), '# Requirements\n- TEST-01\n');

  // CONTEXT.md
  fs.writeFileSync(path.join(phaseDir, '01-CONTEXT.md'), '# Context\nTest context\n');

  if (opts.agentSpec) {
    fs.writeFileSync(path.join(phaseDir, '01-AGENT-SPEC.md'), '---\nstatus: draft\n---\n# Agent Spec\n');
  }

  return { planningDir, phaseDir };
}

describe('AGENT-SPEC discovery in init plan-phase', () => {
  let tmpDir;

  beforeEach(() => {
    tmpDir = fs.mkdtempSync(path.join(os.tmpdir(), 'gsd-agent-spec-test-'));
  });

  afterEach(() => {
    fs.rmSync(tmpDir, { recursive: true, force: true });
  });

  test('returns has_agent_spec: true when AGENT-SPEC.md present', () => {
    setupProject(tmpDir, { agentSpec: true });
    const result = runInit(tmpDir);
    expect(result.has_agent_spec).toBe(true);
    expect(result.agent_spec_path).toContain('AGENT-SPEC.md');
  });

  test('returns has_agent_spec: false when no AGENT-SPEC.md', () => {
    setupProject(tmpDir, { agentSpec: false });
    const result = runInit(tmpDir);
    expect(result.has_agent_spec).toBe(false);
    expect(result.agent_spec_path).toBeUndefined();
  });

  test('agent_spec_path uses posix separators', () => {
    setupProject(tmpDir, { agentSpec: true });
    const result = runInit(tmpDir);
    if (result.agent_spec_path) {
      expect(result.agent_spec_path).not.toContain('\\\\');
      expect(result.agent_spec_path).toContain('/');
    }
  });
});
```
